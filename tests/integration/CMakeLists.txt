cmake_minimum_required(VERSION 3.24)
project(strata_integration_contracts LANGUAGES CXX CUDA)

# Standalone diagnostics: no change to the production build/default dispatch.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 87)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(CUDAToolkit REQUIRED)
get_filename_component(STRATA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(weighted_rms_contract
  weighted_rms_contract.cpp
  ${STRATA_ROOT}/src/kernels/cuda/native_gr_norm.cu)
target_include_directories(weighted_rms_contract PRIVATE ${STRATA_ROOT}/include)
target_compile_options(weighted_rms_contract PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math>)
target_link_libraries(weighted_rms_contract PRIVATE CUDA::cudart)
enable_testing()
add_test(NAME weighted_rms_contract COMMAND weighted_rms_contract)
set_tests_properties(weighted_rms_contract PROPERTIES SKIP_RETURN_CODE 77)
set(STRATA_GGML_DIR "" CACHE PATH "Pinned llama.cpp source for codec diagnostics; no fetching")
if(STRATA_GGML_DIR)
  if(NOT EXISTS "${STRATA_GGML_DIR}/ggml/src/ggml-common.h")
    message(FATAL_ERROR "STRATA_GGML_DIR must contain the pinned ggml sources")
  endif()
  add_library(integration_codec_kernels STATIC
    ${STRATA_ROOT}/src/kernels/cuda/dequant_bf16.cu
    ${STRATA_ROOT}/src/kernels/cuda/iq_kernels.cu)
  target_include_directories(integration_codec_kernels PRIVATE ${STRATA_ROOT}/include
    ${STRATA_GGML_DIR}/ggml/src)
  target_compile_options(integration_codec_kernels PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math>)
  target_link_libraries(integration_codec_kernels PUBLIC CUDA::cudart)
  foreach(codec IN ITEMS iq4_nl q6_k)
    add_executable(${codec}_codec_contract ${codec}_codec_contract.cpp)
    target_include_directories(${codec}_codec_contract PRIVATE ${STRATA_ROOT}/include)
    target_link_libraries(${codec}_codec_contract PRIVATE integration_codec_kernels)
    add_test(NAME ${codec}_codec_contract COMMAND ${codec}_codec_contract)
    set_tests_properties(${codec}_codec_contract PROPERTIES SKIP_RETURN_CODE 77)
  endforeach()
  add_executable(q6_k_real_payload_contract q6_k_real_payload_contract.cpp)
  target_include_directories(q6_k_real_payload_contract PRIVATE ${STRATA_ROOT}/include)
  target_link_libraries(q6_k_real_payload_contract PRIVATE integration_codec_kernels CUDA::cudart)
  add_executable(q6_k_real_payload_bench q6_k_real_payload_bench.cpp)
  target_include_directories(q6_k_real_payload_bench PRIVATE ${STRATA_ROOT}/include)
  target_link_libraries(q6_k_real_payload_bench PRIVATE integration_codec_kernels CUDA::cudart)
else()
  message(STATUS "Codec diagnostics not configured: supply pinned STRATA_GGML_DIR")
endif()
